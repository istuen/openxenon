<script setup lang="ts">
import { onMounted, ref } from 'vue'

interface Health {
  generatedAt: string
  summary: { totalWorks: number; activeWorks: number; passedRounds: number; failedRounds: number }
  worksByStatus: Record<string, number>
  commonFailurePatterns: { pattern: string; count: number }[]
}

const health = ref<Health | null>(null)
const error = ref<string | null>(null)

onMounted(async () => {
  try {
    const res = await fetch('/hall/health.json', { cache: 'no-store' })
    if (!res.ok) throw new Error(`HTTP ${res.status}`)
    health.value = await res.json()
  } catch (e) {
    error.value = (e as Error).message
  }
})
</script>

<template>
  <div class="hall-health">
    <h1>OXN Health Dashboard</h1>
    <p v-if="error" class="error">Failed to load health data: {{ error }}</p>
    <template v-else-if="health">
      <div class="metrics">
        <MetricCard label="Total Works" :value="health.summary.totalWorks" />
        <MetricCard label="Active" :value="health.summary.activeWorks" />
        <MetricCard label="Passed" :value="health.summary.passedRounds" />
        <MetricCard label="Failed" :value="health.summary.failedRounds" />
      </div>
      <h2>Works by Status</h2>
      <ul>
        <li v-for="(count, status) in health.worksByStatus" :key="status">
          <code>{{ status }}</code>: {{ count }}
        </li>
      </ul>
      <h2>Top Failure Patterns</h2>

### 5.3 check-docs-links 脚本

```typescript
// scripts/check-docs-links.ts (L3 工具脚本)
import { readdir, readFile } from 'node:fs/promises'
import { join } from 'node:path'

const DOCS_DIRS = ['docs/zh-cn', 'docs/en', 'README.md', 'README.en.md']

async function checkLinks(file: string): Promise<string[]> {
  const content = await readFile(file, 'utf8')
  const errors: string[] = []
  const linkRegex = /\[([^\]]+)\]\(([^)]+)\)/g
  let match
  
  while ((match = linkRegex.exec(content)) !== null) {
    const [, text, href] = match
    if (href.startsWith('http') || href.startsWith('#')) continue
    if (href.startsWith('/')) {
      // 绝对路径，检查文件存在
      const path = join('docs', href)
      try {
        await readFile(path, 'utf8')
      } catch {
        errors.push(`${file}: broken link [${text}](${href})`)
      }
    }
  }
  
  return errors
}

async function main() {
  let allErrors: string[] = []
  for (const dir of DOCS_DIRS) {
    // 递归扫描
    const files = await scanMarkdownFiles(dir)
    for (const file of files) {
      const errors = await checkLinks(file)
      allErrors = allErrors.concat(errors)
    }
  }
  
  if (allErrors.length > 0) {
    console.error('❌ Broken links:')
    allErrors.forEach(e => console.error(`  ${e}`))
    process.exit(1)
  }
  console.log('✓ All links valid')
}
```
